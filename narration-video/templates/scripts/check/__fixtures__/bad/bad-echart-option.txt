// 违反「ECharts option 裸色/裸字体」（Plan 4.5 门禁新条目，blocker）：
//   - option 内 color: "#f00" 裸 hex 色值（色须走注册主题——章节层 option 不写颜色字面量）；
//   - option 内 textStyle.fontFamily "Arial" 裸字体（须走 var(--font-*)）。
export default function BadChart({ step }: { step: number }) {
  return (
    <EChart
      option={{
        color: "#f00",
        textStyle: { fontFamily: "Arial" },
        series: [{ type: "bar", data: [step] }],
      }}
    />
  );
}
